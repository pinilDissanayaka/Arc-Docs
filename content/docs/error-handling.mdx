azcore provides a comprehensive exception hierarchy and error handling utilities to build robust, production-ready multi-agent systems. Proper error handling ensures graceful degradation, informative debugging, and reliable operation.

## Exception Hierarchy

### AzCoreException Base Class

All azcore exceptions inherit from `AzCoreException`:

```python
from azcore.exceptions import AzCoreException

class AzCoreException(Exception):
"""
Base exception for all azcore errors.

Attributes:
message: Human-readable error message
details: Dictionary with additional error context
"""

def __init__(self, message: str, details: dict = None):
super().__init__(message)
self.message = message
self.details = details or {}
```

## Exception Types

### Configuration Errors

```python

# Raised when configuration is invalid
```

- Required configuration keys missing
- Configuration values out of valid range
- Environment variables not set

## Error Handling Patterns

### Try-Except with Specific Exceptions

```python
from azcore.exceptions import (
LLMError,
NodeExecutionError,
StateError
)

def safe_node_execution(node, state):
"""Execute node with comprehensive error handling."""

try:
result = node.execute(state)
return result

except StateError as e:
# State issue - can't proceed
logger.error(f"State error: {e.message}")
logger.error(f"Details: {e.details}")
raise # Re-raise, can't recover

except LLMError as e:
# LLM issue - might be temporary
logger.warning(f"LLM error: {e.message}")
# Retry with backoff
return retry_with_backoff(node, state)

except NodeExecutionError as e:
# Node failed - use fallback
logger.error(f"Node execution failed: {e.message}")
return fallback_node.execute(state)

except AzCoreException as e:
# Catch any azcore exception
logger.error(f"azcore error: {e.message}")
return error_recovery(e)

except Exception as e:
# Unexpected error
logger.error(f"Unexpected error: {e}")
raise
```

### Retry Decorator

```python
from azcore.utils.retry import retry_with_timeout

class RobustNode(BaseNode):
"""Node with automatic retries."""

@retry_with_timeout(max_retries=3, timeout=30.0)
def execute(self, state):
"""Execute with retries and timeout."""

# Validate
if not state or "messages" not in state:
raise ValidationError("Invalid state structure")

# Process
result = self._process(state)

return Command(
update={"messages": [result]},
goto="next"
)

# Will retry up to 3 times on failure
# Will timeout after 30 seconds
```

### Error Recovery Nodes

```python
def create_error_handler_node(fallback_message: str):
"""Create a node that handles errors."""

def error_handler(state):
"""Handle errors and provide fallback."""

error = state.get("metadata", {}).get("error")
error_details = state.get("metadata", {}).get("error_details", {})

# Log error
logger.error(f"Error in workflow: {error}")
logger.error(f"Details: {error_details}")

# Determine recovery strategy
if isinstance(error, LLMError):
message = "LLM service temporarily unavailable. Using cached response."
goto = "cache_fallback"
elif isinstance(error, ToolExecutionError):
message = "Tool execution failed. Using alternative approach."
goto = "alternative_tool"
else:
message = fallback_message
goto = END

return Command(
update={
"messages": [
HumanMessage(content=message, name="error_handler")
],
"metadata": {
**state.get("metadata", {}),
"error_handled": True
}
},
goto=goto
)

return error_handler

# Add to graph
error_handler = create_error_handler_node(
fallback_message="An error occurred. Please try again."
)
orchestrator.add_node("error_handler", error_handler)
```

### Graceful Degradation

```python
class DegradableNode(BaseNode):
"""Node that degrades gracefully on errors."""

def execute(self, state):
"""Execute with fallback strategies."""

# Try primary strategy
try:
return self._primary_strategy(state)
except LLMError:
logger.warning("Primary strategy failed, trying secondary")

# Try secondary strategy
try:
return self._secondary_strategy(state)
except Exception:
logger.warning("Secondary strategy failed, using minimal")

# Minimal fallback
return self._minimal_fallback(state)

def _primary_strategy(self, state):
"""Full-featured processing with LLM."""
result = self.llm.invoke(state["messages"])
return Command(
update={"messages": [result]},
goto="next"
)

def _secondary_strategy(self, state):
"""Cached or rule-based processing."""
cached = self.cache.get(state["messages"][-1].content)
if cached:
return Command(
update={"messages": [cached]},
goto="next"
)
raise Exception("No cache available")

def _minimal_fallback(self, state):
"""Absolute minimum functionality."""
return Command(
update={
"messages": [
HumanMessage(
content="Service temporarily limited. Please try again.",
name=self.name
)
]
},
goto=END
)
```

## Error Recovery Strategies

### Retry with Exponential Backoff

```python
import time
from typing import Callable, Any

def retry_with_exponential_backoff(
func: Callable,
max_retries: int = 3,
base_delay: float = 1.0,
max_delay: float = 60.0
) -> Any:
"""Retry with exponential backoff."""

for attempt in range(max_retries):
try:
return func()

except (LLMError, ToolExecutionError) as e:
if attempt == max_retries - 1:
raise

# Calculate delay
delay = min(base_delay * (2 ** attempt), max_delay)

logger.warning(
f"Attempt {attempt + 1} failed: {e.message}. "
f"Retrying in {delay}s..."
)

time.sleep(delay)

raise Exception("Max retries exceeded")

# Use with nodes
def resilient_node(state):
def process():
return llm.invoke(state["messages"])

result = retry_with_exponential_backoff(process)
return Command(update={"messages": [result]}, goto="next")
```

### Circuit Breaker

```python
from datetime import datetime, timedelta

class CircuitBreaker:
"""Circuit breaker pattern for error handling."""

def __init__(
self,
failure_threshold: int = 5,
recovery_timeout: float = 60.0
):
self.failure_threshold = failure_threshold
self.recovery_timeout = recovery_timeout
self.failures = 0
self.last_failure_time = None
self.state = "closed" # closed, open, half_open

def call(self, func: Callable) -> Any:
"""Execute function with circuit breaker."""

# Check if circuit should reset
if self.state == "open":
if datetime.now() - self.last_failure_time > timedelta(
seconds=self.recovery_timeout
):
self.state = "half_open"
self.failures = 0

# Prevent calls when circuit is open
if self.state == "open":
raise Exception("Circuit breaker is open")

try:
result = func()

# Success - reset failures
if self.state == "half_open":
self.state = "closed"
self.failures = 0

return result

except Exception as e:
self.failures += 1
self.last_failure_time = datetime.now()

if self.failures >= self.failure_threshold:
self.state = "open"
logger.error("Circuit breaker opened due to failures")

raise

# Use with LLM calls
llm_breaker = CircuitBreaker(failure_threshold=3)

def protected_llm_call(messages):
return llm_breaker.call(lambda: llm.invoke(messages))
```

### Fallback Chain

```python
class FallbackChain:
"""Try multiple strategies in order."""

def __init__(self, strategies: list[Callable]):
self.strategies = strategies

def execute(self, state):
"""Try each strategy until one succeeds."""

errors = []

for i, strategy in enumerate(self.strategies):
try:
logger.info(f"Trying strategy {i + 1}/{len(self.strategies)}")
return strategy(state)

except Exception as e:
errors.append(str(e))
logger.warning(f"Strategy {i + 1} failed: {e}")

if i == len(self.strategies) - 1:
# All strategies failed
raise Exception(
f"All fallback strategies failed: {errors}"
)

raise Exception("No strategies available")

# Use in nodes
def primary_strategy(state):
"""Primary processing with full features."""
result = expensive_llm.invoke(state["messages"])
return Command(update={"messages": [result]}, goto="next")

def secondary_strategy(state):
"""Secondary processing with cheaper model."""
result = cheap_llm.invoke(state["messages"])
return Command(update={"messages": [result]}, goto="next")

def tertiary_strategy(state):
"""Tertiary rule-based processing."""
result = rule_based_processor(state["messages"][-1].content)
return Command(
update={"messages": [HumanMessage(content=result)]},
goto="next"
)

# Create fallback chain
fallback = FallbackChain([
primary_strategy,
secondary_strategy,
tertiary_strategy
])

# Use in node
def resilient_node(state):
return fallback.execute(state)
```

## Best Practices

### 1. Catch Specific Exceptions

```python
# GOOD: Catch specific exceptions
try:
result = node.execute(state)
except LLMError as e:
# Handle LLM error specifically
pass
except NodeExecutionError as e:
# Handle node error specifically
pass

# BAD: Catch all exceptions
try:
result = node.execute(state)
except Exception:
# Too broad, hides errors
pass
```

### 2. Provide Context in Exceptions

```python
# GOOD: Include helpful context
raise NodeExecutionError(
"Failed to process user request",
details={
"node": self.name,
"user_id": state.get("context", {}).get("user_id"),
"message_count": len(state["messages"]),
"error": str(e)
}
)

# BAD: Generic error
raise Exception("Error")
```

### 3. Log Before Re-raising

```python
# GOOD: Log then re-raise
try:
result = risky_operation()
except Exception as e:
logger.error(f"Operation failed: {e}", exc_info=True)
raise

# Provides logging even when exception propagates
```

### 4. Use Retry Decorators

```python
# GOOD: Use built-in retry utilities
from azcore.utils.retry import retry_with_timeout

@retry_with_timeout(max_retries=3, timeout=30.0)
def execute(self, state):
return self._process(state)

# BAD: Manual retry logic everywhere
def execute(self, state):
for i in range(3):
try:
return self._process(state)
except:
if i == 2:
raise
time.sleep(1)
```

### 5. Design for Failure

```python
# GOOD: Assume operations might fail
def robust_node(state):
"""Node designed for failure scenarios."""

# Validate inputs
if not validate_state(state):
return error_response("Invalid state")

# Try operation with fallback
try:
result = primary_operation(state)
except Exception as e:
logger.warning(f"Primary failed: {e}")
result = fallback_operation(state)

# Validate outputs
if not validate_result(result):
return error_response("Invalid result")

return result
```

## Complete Error Handling Example

```python
from azcore.core.base import BaseNode
from azcore.exceptions import (
NodeExecutionError,
ValidationError,
LLMError,
ToolExecutionError
)
from azcore.utils.retry import retry_with_timeout
from langgraph.types import Command
from langchain_core.messages import HumanMessage
import logging

logger = logging.getLogger(__name__)

class ProductionNode(BaseNode):
"""Production-ready node with comprehensive error handling."""

def __init__(self, llm, tools, name="production_node"):
super().__init__(name=name, description="Production node")
self.llm = llm
self.tools = tools
self.circuit_breaker = CircuitBreaker(failure_threshold=3)

@retry_with_timeout(max_retries=3, timeout=30.0)
def execute(self, state):
"""Execute with full error handling."""

try:
# Validate state
self._validate_state(state)

# Process with circuit breaker
result = self.circuit_breaker.call(
lambda: self._process(state)
)

return Command(
update={
"messages": [result],
"metadata": {
**state.get("metadata", {}),
"success": True
}
},
goto="next_node"
)

except ValidationError as e:
logger.error(f"Validation error: {e.message}")
return self._validation_error_response(e)

except LLMError as e:
logger.error(f"LLM error: {e.message}")
return self._llm_error_response(state, e)

except ToolExecutionError as e:
logger.error(f"Tool error: {e.message}")
return self._tool_error_response(state, e)

except Exception as e:
logger.error(f"Unexpected error: {e}", exc_info=True)
raise NodeExecutionError(
f"Node {self.name} failed unexpectedly",
details={
"node": self.name,
"error": str(e),
"error_type": type(e).__name__
}
)

def _validate_state(self, state):
"""Validate state structure."""
required_fields = ["messages"]

for field in required_fields:
if field not in state:
raise ValidationError(
f"Missing required field: {field}",
details={"field": field, "state_keys": list(state.keys())}
)

def _process(self, state):
"""Core processing logic."""
messages = state["messages"]

# Invoke LLM with tools
response = self.llm.invoke(messages)

return HumanMessage(content=response.content, name=self.name)

def _validation_error_response(self, error):
"""Handle validation errors."""
return Command(
update={
"messages": [
HumanMessage(
content="Invalid input. Please check your request.",
name=self.name
)
],
"metadata": {"error": "validation", "details": error.details}
},
goto="error_handler"
)

def _llm_error_response(self, state, error):
"""Handle LLM errors with fallback."""
# Try cached response
cached = self._get_cached_response(state)
if cached:
return Command(
update={
"messages": [cached],
"metadata": {"used_cache": True}
},
goto="next_node"
)

# No cache, return error
return Command(
update={
"messages": [
HumanMessage(
content="Service temporarily unavailable. Please try again.",
name=self.name
)
],
"metadata": {"error": "llm_unavailable"}
},
goto=END
)

def _tool_error_response(self, state, error):
"""Handle tool errors."""
return Command(
update={
"messages": [
HumanMessage(
content=f"Unable to complete action: {error.message}",
name=self.name
)
],
"metadata": {
"error": "tool_execution",
"tool": error.details.get("tool")
}
},
goto="error_handler"
)
```
